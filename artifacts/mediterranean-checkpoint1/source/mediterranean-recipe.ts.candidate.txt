/** Synthetic descending garrigue shoulder, not claimed measured topography. */
export function mediterraneanHeight(seed:string,x:number,z:number){
 const phase=hashSeed(seed+'|garrigue-landform')/4294967296*6.283;
 const h=(a:number,b:number)=>{
  const d=Math.max(0,16-b),fall=-10.5*(1-Math.exp(-d/48));
  const shoulders=3.9*(1-Math.exp(-Math.pow(a/18,2)))*Math.exp(-d/420);
  const right=2.0*Math.exp(-Math.pow((a-10)/12,2)-Math.pow((b+5)/26,2));
  const middle=5.5*Math.exp(-Math.pow((a+55)/50,2)-Math.pow((b+100)/70,2))+7*Math.exp(-Math.pow((a-110)/95,2)-Math.pow((b+205)/95,2));
  const rolls=.44*Math.sin(a*.12+phase)*Math.cos(b*.085)+.22*Math.sin(b*.21+a*.17+phase);
  return fall+shoulders+right+middle+rolls;
 };return h(x,z)-h(0,16);
}
export function mediterraneanCommunity(seed:string,x:number,z:number){const p=hashSeed(seed+'|garrigue-communities')/4294967296*6.283;return T.MathUtils.clamp(.50+.25*Math.sin(x*.12+Math.sin(z*.065+p)*2)+.18*Math.sin(z*.17+x*.055+p),0,1);}
/** Sample the reference only while constructing a fixed world field. Never follows the live camera. */
export function referenceSoilSampler(seed:string,mask:{width:number;height:number;pixels:number[]}){
 const camera=new T.PerspectiveCamera(47,16/9,.2,2500);camera.position.fromArray(mediterraneanReferenceCamera.position);camera.lookAt(...mediterraneanReferenceCamera.target);camera.updateMatrixWorld();const point=new T.Vector3();
 return (x:number,z:number)=>{point.set(x,mediterraneanHeight(seed,x,z),z).project(camera);const u=(point.x+1)*.5,v=(1-point.y)*.5;
  if(point.z>=-1&&point.z<=1&&u>=0&&u<=1&&v>=0&&v<=1){const px=u*(mask.width-1),py=v*(mask.height-1),ix=Math.floor(px),iy=Math.floor(py),fx=px-ix,fy=py-iy,at=(a:number,b:number)=>mask.pixels[Math.min(mask.height-1,b)*mask.width+Math.min(mask.width-1,a)]/255;return (at(ix,iy)*(1-fx)+at(ix+1,iy)*fx)*(1-fy)+(at(ix,iy+1)*(1-fx)+at(ix+1,iy+1)*fx)*fy;}
  const center=Math.sin(z*.04)*1.9;return (1-T.MathUtils.smoothstep(Math.abs(x-center),1.1,2.0))*(1-T.MathUtils.smoothstep(Math.abs(z),90,160));
 };
}
function shapeOak(source:SceneTree,height:number,radius:number,index:number,seed:string){
 const study=temperateArchitecture(source,height,radius,index,seed),hero=index===0;
 const transform=(p:Point):Point=>[p[0]+(hero?4.8:.35)*Math.pow(Math.max(0,p[1]/height),1.15),p[1],p[2]];
 const tangent=(v:Point|undefined,y:number):Point=>{const t=new T.Vector3(...(v??[0,1,0]));t.x+=(hero?4.8:.35)*1.15*Math.pow(Math.max(0,y/height),.15)/height*t.y;return t.normalize().toArray() as Point;};
 const segments=study.segments.map(s=>({...s,a:transform(s.a),b:transform(s.b),r0:s.r0*(hero?1.36:1),r1:s.r1*(hero?1.36:1),color:'#614c35',tangentA:tangent(s.tangentA,s.a[1]),tangentB:tangent(s.tangentB,s.b[1])}));
 // Follow the displaced branch skeleton; keep optical leaf sizes and branch ownership.
 const fronds=study.fronds.filter((f,i)=>hero||index<4||i%3===0).map(f=>({...f,position:transform(f.position),width:f.width*.86,length:f.length*.86}));
 const groups=study.groups.map(g=>({...g,center:transform(g.center),anchor:transform(g.anchor)}));
 const model={...study.model,segments,dimensions:{height:height+1,crownWidth:radius*2+(hero?5:1),crownDepth:radius*2+2},palette:{name:'Mediterranean candidate oak',shadow:'#344533',body:'#747d3e',light:'#b8ba67',bark:'#614c35'}};
 return {...study,segments,fronds,groups,model};
}

 const source=r.trees.filter(t=>t.species.scientificName==='Quercus ilex'),rng=randomSource(r.seed+'|mediterranean-occurrences'),trees:SceneTree[]=[];
 // Relative roles, not copied concept coordinates; the seed perturbs spacing, size and yaw.
 const roles=[[-3.8,9,9.2,5.8],[11,-12,6.4,4.0],[-14,-35,4.8,3.4],[19,-55,5.1,3.8],[-38,-95,4.7,3.4],[44,-125,5.3,4.0],[-70,-165,5.4,3.9],[93,-235,6.0,4.4],[10,-115,3.7,2.7],[-102,-275,6,4.2],[135,-340,7,4.6]];
 for(let i=0;i<roles.length&&source.length;i++){const t=source[i%source.length],[a,b,h,w]=roles[i],x=a+(rng()-.5)*(i<2?.5:5),z=b+(rng()-.5)*(i<2?.6:8),studio=shapeOak(t,h*(.94+rng()*.12),w*(.94+rng()*.12),i,r.seed);trees.push({...t,id:t.id+'-mediterranean-presentation-'+i,index:i,position:[x,mediterraneanHeight(r.seed,x,z),z],rotation:i===0?0:rng()*6.283,studio,model:studio.model,species:{...t.species,visual:{...t.species.visual,fallback:true,reason:'Regional Quercus ilex candidate; shared oak branch/spray approximation, not a measured species reconstruction or point occurrence.'}}});}
 return {...r,trees,purpose:'visual_laboratory',distribution:'Reference-role composition with deterministic variations using existing candidate oak individuals; no measured tree positions or abundance.',limitations:[...r.limitations,'Garrigue terrain, oak size and unnamed shrub communities are synthetic. Concept-matching layout is not a geographic reconstruction.']};
}
