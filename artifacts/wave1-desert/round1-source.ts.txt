import * as T from 'three';
import {randomSource,hashSeed} from './random';
import type {SceneRecipe} from './recipe';
import type {CameraView} from './render-types';
const phase=(seed:string)=>(hashSeed(seed+'|desert-relief')%6283)/1000;
export function desertHeight(seed:string,x:number,z:number){const p=phase(seed),wash=x-7*Math.sin(z*.016+p);return .012*(34-z)+.65*Math.sin(x*.035+p)*Math.cos(z*.018)-1.1*Math.exp(-wash*wash/55)+Math.max(0,Math.sin((x+z*.12)*.009+p))**2*2;}
function camera(recipe:SceneRecipe):CameraView{return{position:[0,desertHeight(recipe.seed,0,34)+1.7,34],target:[1,desertHeight(recipe.seed,1,-90)+3,-90]};}
function compose(recipe:SceneRecipe):SceneRecipe{return{...recipe,distribution:recipe.distribution+'; Desert visual study keeps original ecological trees, if any. Nonwoody dressing follows unnamed resolver xeric-shrub form.'};}
function rockGeometry(seed:number){const g=new T.IcosahedronGeometry(1,3),a=g.getAttribute('position');for(let i=0;i<a.count;i++){const x=a.getX(i),y=a.getY(i),z=a.getZ(i),weather=1+.12*Math.sin(x*7+y*3+seed)+.065*Math.sin(z*11-y*4+seed);a.setXYZ(i,x*weather,y*weather*.8,z*weather);}g.computeVertexNormals();return g;}
function shrubGeometry(){const pos:number[]=[],color:number[]=[],idx:number[]=[];const r=randomSource('xeric-shrub-growth-form');for(let arm=0;arm<9;arm++){const a=arm*2.399,length=.3+r()*.5,x=Math.cos(a)*length,z=Math.sin(a)*length,h=.25+r()*.4,base=pos.length/3;pos.push(0,0,0,x-.015,h,z,x+.015,h,z);color.push(.25,.23,.16,.31,.33,.23,.31,.33,.23);idx.push(base,base+1,base+2);for(let lobe=0;lobe<2;lobe++){const start=pos.length/3,y=h+lobe*.11,rad=.16+r()*.09;pos.push(x,y+.1,z);color.push(.37,.41,.28);for(let j=0;j<7;j++){const b=j/6*Math.PI*2;pos.push(x+Math.cos(b)*rad,y-.08,z+Math.sin(b)*rad);color.push(.27,.32,.23);}for(let j=0;j<6;j++)idx.push(start,start+1+j,start+2+j);}}const g=new T.BufferGeometry();g.setAttribute('position',new T.Float32BufferAttribute(pos,3));g.setAttribute('color',new T.Float32BufferAttribute(color,3));g.setIndex(idx);g.computeVertexNormals();return g;}
function mount(scene:T.Scene,recipe:SceneRecipe,sun:T.DirectionalLight,fill:T.DirectionalLight){const seed=recipe.seed,r=randomSource(seed+'|desert-composition-1'),p=phase(seed),textures:T.Texture[]=[],o=new T.Object3D();for(const child of scene.children)if(child instanceof T.Mesh&&['PlaneGeometry','BoxGeometry'].includes(child.geometry.type))child.visible=false;
 scene.userData.labAssetsReady=false;const skyMap=new T.TextureLoader().load('/assets/puget-v1/sky-matte-v1.webp',()=>{scene.userData.labAssetsReady=true;scene.userData.artRevision=(scene.userData.artRevision??0)+1;},undefined,()=>{scene.userData.labAssetsReady=true;scene.userData.wave1AssetError='Sky asset unavailable';});skyMap.colorSpace=T.SRGBColorSpace;textures.push(skyMap);const sky=new T.Mesh(new T.SphereGeometry(1800,40,20),new T.MeshBasicMaterial({map:skyMap,side:T.BackSide,color:'#e7dcce',fog:false,depthWrite:false,toneMapped:false}));sky.name='desert-shared-sky';sky.renderOrder=-10;scene.add(sky);
 const terrainMat=new T.MeshStandardMaterial({vertexColors:true,roughness:1});terrainMat.onBeforeCompile=s=>{s.vertexShader='varying vec3 desertWorld;\n'+s.vertexShader;s.vertexShader=s.vertexShader.replace('#include <begin_vertex>','#include <begin_vertex>\ndesertWorld=(modelMatrix*vec4(position,1.)).xyz;');s.fragmentShader='varying vec3 desertWorld;\n'+s.fragmentShader;s.fragmentShader=s.fragmentShader.replace('#include <color_fragment>','#include <color_fragment>\nfloat grit=sin(desertWorld.x*18.+sin(desertWorld.z*9.))*sin(desertWorld.z*23.);diffuseColor.rgb*=1.+grit*.035*(1.-smoothstep(10.,65.,distance(cameraPosition,desertWorld)));');};
 const land=new T.PlaneGeometry(1800,1800,160,160);land.rotateX(-Math.PI/2);land.translate(0,0,-700);const a=land.getAttribute('position'),col=new Float32Array(a.count*3);for(let i=0;i<a.count;i++){const x=a.getX(i),z=a.getZ(i),wash=x-7*Math.sin(z*.016+p),y=desertHeight(seed,x,z),c=new T.Color('#b79b70').lerp(new T.Color('#c8b994'),Math.exp(-wash*wash/85)*.8).lerp(new T.Color('#8e846c'),(.5+.5*Math.sin(x*.039+z*.021+p))*.25);a.setY(i,y);col.set(c.toArray(),i*3);}land.setAttribute('color',new T.BufferAttribute(col,3));land.computeVertexNormals();const terrain=new T.Mesh(land,terrainMat);terrain.receiveShadow=true;terrain.name='desert-continuous-alluvial-fan';scene.add(terrain);
 // Actual weathered ridge geometry: broad connected shoulders, broken crests, cooler far layers.
 for(let layer=0;layer<3;layer++){const g=new T.PlaneGeometry(2200,700,128,32);g.rotateX(-Math.PI/2);g.translate(0,0,-500-layer*330);const a=g.getAttribute('position'),c=new Float32Array(a.count*3),peakPhase=p+layer*1.7;for(let i=0;i<a.count;i++){const x=a.getX(i),z=a.getZ(i),u=(z+500+layer*330)/350,envelope=Math.pow(Math.max(0,1-u*u),1.4),ridge=34+36*Math.pow(.5+.5*Math.sin(x*.012+peakPhase),2)+18*Math.abs(Math.sin(x*.029+peakPhase));a.setY(i,desertHeight(seed,x,z)+envelope*ridge);const shade=new T.Color(layer===0?'#958779':layer===1?'#929697':'#a6afb0');shade.multiplyScalar(.87+.15*Math.sin(x*.03+z*.01));c.set(shade.toArray(),i*3);}g.setAttribute('color',new T.BufferAttribute(c,3));g.computeVertexNormals();const m=new T.Mesh(g,new T.MeshStandardMaterial({vertexColors:true,roughness:1}));m.name='desert-eroded-ridge-'+layer;scene.add(m);}
 const rockMat=new T.MeshStandardMaterial({roughness:1});rockMat.onBeforeCompile=s=>{s.vertexShader='varying vec3 rockWorld;\n'+s.vertexShader;s.vertexShader=s.vertexShader.replace('#include <begin_vertex>','#include <begin_vertex>\nrockWorld=(modelMatrix*instanceMatrix*vec4(position,1.)).xyz;');s.fragmentShader='varying vec3 rockWorld;\n'+s.fragmentShader;s.fragmentShader=s.fragmentShader.replace('#include <color_fragment>','#include <color_fragment>\nfloat joint=pow(abs(sin(rockWorld.y*.9+sin(rockWorld.x*.17)*.2)),30.);diffuseColor.rgb*=.9+.1*sin(rockWorld.x*3.+rockWorld.z*4.);diffuseColor.rgb*=1.-joint*.15;');};
 const counts=[38,70,105];let rocks=0;for(let variant=0;variant<3;variant++){const mesh=new T.InstancedMesh(rockGeometry(variant+3),rockMat,counts[variant]);mesh.name='desert-weathered-granite';mesh.castShadow=true;mesh.receiveShadow=true;for(let i=0;i<mesh.count;i++){const cluster=i<14?0:i<29?1:2,side=cluster===0?1:-1,cx=cluster===0?10:cluster===1?-25:35,cz=cluster===0?20:cluster===1?-32:-110,angle=r()*6.283,span=i<29?4+r()*4:10+r()*20,x=cx+Math.cos(angle)*span,z=cz+Math.sin(angle)*span,size=i<5?(variant===0?2.3+r()*1.2:1+r()*1.4):.25+r()*(cluster===1?3:1.5);o.position.set(x,desertHeight(seed,x,z)+size*.28,z);o.rotation.set(r()*.35,r()*6.283,r()*.15);o.scale.set(size*(1+r()*.5),size*(1+r()*.5),size);o.updateMatrix();mesh.setMatrixAt(i,o.matrix);mesh.setColorAt(i,new T.Color('#ab8c69').lerp(new T.Color('#d0b28b'),r()*.5));}mesh.computeBoundingSphere();scene.add(mesh);rocks+=mesh.count;}
 const shrubCount=280,shrubs=new T.InstancedMesh(shrubGeometry(),new T.MeshStandardMaterial({vertexColors:true,roughness:1,side:T.DoubleSide}),shrubCount);shrubs.name='desert-resolver-xeric-shrub-form';for(let i=0;i<shrubCount;i++){let x=0,z=0;for(let k=0;k<12;k++){x=(r()-.5)*200;z=35-r()*280;const wash=Math.abs(x-7*Math.sin(z*.016+p)),clump=.5+.5*Math.sin(x*.11+z*.06);if(wash>3&&r()<clump*.7)break;}const size=.35+r()*.75;o.position.set(x,desertHeight(seed,x,z),z);o.rotation.set(0,r()*6.283,0);o.scale.setScalar(size);o.updateMatrix();shrubs.setMatrixAt(i,o.matrix);}shrubs.computeBoundingSphere();shrubs.receiveShadow=true;scene.add(shrubs);
 sun.position.set(-75,65,35);sun.color.set('#ffe0ad');sun.intensity=3.2;sun.shadow.intensity=.55;fill.color.set('#8c9eac');fill.intensity=.5;scene.children.forEach(c=>{if(c instanceof T.HemisphereLight){c.color.set('#b8ccdd');c.groundColor.set('#ad8c66');c.intensity=.8;}});scene.fog=new T.Fog('#c1c8c5',145,1400);scene.background=new T.Color('#becbd0');scene.userData.wave1={id:'desert',biome:13,seed,rocks,understoryInstances:shrubCount,ecology:'Only unnamed xeric_shrub growth form from resolver; no Joshua-tree presence asserted. Synthetic geology, not local DEM.',season:'not inferred',weather:'clear presentation layer',timeOfDay:'warm oblique lighting presentation layer'};scene.userData.recipeShadowDirty=true;return()=>textures.forEach(t=>t.dispose());}
export const desertStudy={id:'desert',biome:13,compose,camera,mount};
