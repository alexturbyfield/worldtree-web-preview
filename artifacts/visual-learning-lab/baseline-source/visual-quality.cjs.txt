async function pageFor(biome,variant,reload){const key=biome+'|'+variant;let p=pages.get(key);if(!p){p=await browser.newPage({viewport:{width:1600,height:1050}});p.errors=[];p.on('pageerror',e=>p.errors.push(e.message));p.on('console',m=>{if(m.type()==='error')p.errors.push(m.text().slice(0,2000))});await p.addInitScript(()=>{window.__THREE_DEVTOOLS__=new EventTarget();window.__THREE_DEVTOOLS__.addEventListener('observe',e=>{const r=e.detail;if(!r.isWebGLRenderer)return;const f=r.render;r.render=function(s,c){window.__visual={r,s,c};return f.call(this,s,c)}})});pages.set(key,p);console.log('Navigating',key);await p.goto(ORIGIN+'/explore?'+config.scenes[biome]+config.common+'&pretty='+(variant==='historical'?'off':'on'));console.log('Navigated',key);}else if(reload){p.errors=[];await p.reload();}console.log('Waiting for stage',key);await p.locator('[data-testid=forest-stage][data-state=ready]').waitFor({timeout:120000});console.log('Stage ready',key);await p.waitForFunction(()=>!!window.__visual);await p.waitForTimeout(700);return p;}
async function analyze(png,mask,meta,biome,view){
 const a=await sharp(png).removeAlpha().raw().toBuffer({resolveWithObject:true}),b=mask?await sharp(mask).removeAlpha().raw().toBuffer():null,{width:w,height:h}=a.info;let dark=0,veg=0,roi=0;const bands=Array(12).fill(0);
 for(let y=Math.floor(h*.5);y<h;y++)for(let x=0;x<w;x++){roi++;const i=(y*w+x)*3;if(b&&b[i]>200&&b[i+1]>200&&b[i+2]>200){veg++;bands[Math.min(11,Math.floor(x/w*12))]++;if(.2126*a.data[i]+.7152*a.data[i+1]+.0722*a.data[i+2]<25)dark++;}}
 const findings=[],coverage=veg/Math.max(1,roi),black=dark/Math.max(1,veg),add=(severity,defect,measurements)=>findings.push({severity,defect,measurements});
 if(b&&biome==='savanna'&&coverage<.15)add('major','Insufficient projected lower-half vegetation coverage',{coverage,threshold:.15});
 if(b&&black>.2)add(biome==='savanna'?'major':'minor','Near-black pixels overlap vegetation silhouette mask',{fraction:black,threshold:.2,note:'Includes real shadows and possible occluders; visual review must distinguish legitimate shade.'});
 const floor=meta.instances.find(x=>x.name==='rainforest-v5-floor-decals');if(floor&&floor.visible&&floor.count>100)add(meta.quality?.floorDecals?'minor':'major',meta.quality?.floorDecals?'Four-tile leaf atlas reuse still requires visual repetition review':'Repeated single-atlas floor stamps require repetition review',{count:floor.count,tiles:floor.tiles,scaleRange:floor.scaleRange,albedoMask:meta.quality?.floorDecals??null});
 if(view==='inspection'&&meta.backdrop===false&&biome==='rainforest')add('major','Forest continuation disabled at inspection height',{backdropVisible:false});
 if(meta.errors.length)add('critical','Page or shader errors',{errors:meta.errors});
 return {coverage:{available:!!b,region:'lower half',mask:'actual alpha-tested vegetation silhouette, original geometry; isolated from ground and nonvegetation',fraction:b?coverage:null,nearBlackFraction:b?black:null,horizontalBands:b?bands:null},findings,limitations:['Metrics flag potential failures; they cannot establish artistic quality.','Coverage mask omits terrain occlusion; near-black statistics include legitimate shadows.','Faceting, continuity and conspicuous repetition require vision inspection; no RGB-only artistic approval.']};
}
async function capture(request){const {label,tier=1,scenes=['savanna','rainforest','puget'],variant='current',reload=true,masks=tier===1}=request;if(!/^[\w-]+$/.test(label))throw Error('Invalid capture label');const [width,height]=config.tiers[tier],dir=ROOT+'/'+label;fs.mkdirSync(dir,{recursive:true});const report=[];
 for(const biome of scenes){console.log('Opening',biome,variant);const p=await pageFor(biome,variant,reload);console.log('Ready',biome);for(const view of biome==='puget'?['hero']:['hero','inspection']){
  const d=await p.evaluate(({width,height,view,inspection,masks})=>{const {r,s,c}=window.__visual;window.__visualHero??={position:c.position.toArray(),quaternion:c.quaternion.toArray()};const hero=window.__visualHero;c.position.fromArray(hero.position);c.quaternion.fromArray(hero.quaternion);if(view==='inspection'){c.position.fromArray(inspection.position);c.lookAt(...inspection.target);}r.setPixelRatio(1);r.setSize(width,height,false);c.aspect=width/height;c.updateProjectionMatrix();c.updateMatrixWorld();const times=[];for(let i=0;i<3;i++){const start=performance.now();r.render(s,c);times.push(performance.now()-start);}const png=r.domElement.toDataURL(),performanceInfo={calls:r.info.render.calls,triangles:r.info.render.triangles,cpuSubmissionMs:times,pixelRatio:r.getPixelRatio()},instances=[];
  s.traverse(o=>{if(o.isInstancedMesh){const m=o.instanceMatrix.array,n=o.geometry.getAttribute('prettyTile')||o.geometry.getAttribute('forestTile');let lo=Infinity,hi=0;for(let i=0;i<o.count;i++){const k=i*16,sy=Math.hypot(m[k+4],m[k+5],m[k+6]);lo=Math.min(lo,sy);hi=Math.max(hi,sy);}instances.push({name:o.name,count:o.count,visible:o.visible&&o.parent.visible,tiles:n?new Set(n.array).size:null,scaleRange:[lo,hi]});}});
  let mask=null;if(masks){const visibility=[],materials=[],clones=[],background=s.background,fog=s.fog;
   try{s.background=background?.clone?.().set('#000000')??null;s.fog=null;s.traverse(o=>{if(!o.isMesh)return;visibility.push([o,o.visible]);const vegetation=/grass|ferns|palms|seedlings|shrubs|shrub|foliage-cards/.test(o.name);if(!vegetation){o.visible=false;return;}const maskMaterial=mat=>{const clone=mat.clone(),previous=mat.onBeforeCompile,key=mat.customProgramCacheKey.bind(mat);clone.fog=false;clone.onBeforeCompile=function(shader,renderer){previous.call(this,shader,renderer);if(clone.isShaderMaterial){const end=shader.fragmentShader.lastIndexOf('}');shader.fragmentShader=shader.fragmentShader.slice(0,end)+'gl_FragColor.rgb=vec3(1.);\n'+shader.fragmentShader.slice(end);}else shader.fragmentShader=shader.fragmentShader.replace('#include <dithering_fragment>','#include <dithering_fragment>\ngl_FragColor.rgb=vec3(1.);');};clone.customProgramCacheKey=()=>key()+'|quality-mask';clones.push(clone);return clone;};materials.push([o,o.material]);o.material=Array.isArray(o.material)?o.material.map(maskMaterial):maskMaterial(o.material);});r.render(s,c);mask=r.domElement.toDataURL();}finally{visibility.forEach(([o,v])=>o.visible=v);materials.forEach(([o,m])=>o.material=m);clones.forEach(m=>m.dispose());s.background=background;s.fog=fog;}}
  return {png,mask,camera:{position:c.position.toArray(),quaternion:c.quaternion.toArray(),aspect:c.aspect,fov:c.fov},performance:performanceInfo,instances,backdrop:s.getObjectByName('rainforest-v5-encounter-backdrop')?.visible,plan:s.userData.prettyScene,quality:s.userData.visualQuality,renderer:r.getContext().getParameter(r.getContext().RENDERER)};
 },{width,height,view,inspection:config.inspection,masks});const decode=s=>Buffer.from(s.split(',')[1],'base64'),png=decode(d.png),mask=d.mask?decode(d.mask):null;delete d.png;delete d.mask;d.errors=[...p.errors];const checks=await analyze(png,mask,d,biome,view);const name=biome+'-'+view;fs.writeFileSync(dir+'/'+name+'.png',png);if(mask)fs.writeFileSync(dir+'/'+name+'-mask.png',mask);report.push({biome,view,tier,width,height,variant,...d,...checks});console.log(label,name,d.performance.calls,d.performance.triangles,checks.findings.map(x=>x.defect));}}
 fs.writeFileSync(dir+'/report.json',JSON.stringify(report,null,2));return {label,report};}
