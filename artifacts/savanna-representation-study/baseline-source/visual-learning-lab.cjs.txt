function validateCatalog(data){
 for(const [id,spec] of Object.entries(data.experiments)){
  if(!['A','B'].includes(id)||spec.candidates.length!==3||new Set(spec.candidates.map(c=>c.id)).size!==3)throw Error('Exactly three unique candidates per supported experiment are required');
  const allowed=id==='A'?['geometry','count','blades','segments','bladeWidth','bend','height','heightVariation','width','broadFraction','densityFloor','normalWrap','tip','base']:['spread','fernScale','palmScale','broadleafScale','heightVariation','foregroundScale','litterScale','gap'];
  for(const c of spec.candidates){if(Object.keys(c.parameters).some(k=>!allowed.includes(k)))throw Error('Only the targeted parameter family may vary');const p=c.parameters;if(id==='A'&&(!['atlas','ribbons'].includes(p.geometry)||p.count<=0||p.count>16000||(p.geometry==='atlas'?4*p.count:2*p.blades*p.segments*p.count)>120000))throw Error('Grass candidate exceeds the bounded geometry budget');}
 }return true;
}

function canContinue(s){if(s.visionAvailable===false)return {allowed:false,reason:'Vision inspection unavailable; enable a genuine reviewer before continuing'};if(s.rounds.length>=s.maxRounds)return {allowed:false,reason:'Three-round execution budget exhausted'};const last=s.rounds.slice(-2);if(last.length===2&&last.every(r=>r.meaningfulImprovement===false))return {allowed:false,reason:'Two rounds without meaningful improvement'};if(s.rounds.some(r=>r.status==='awaiting-review'))return {allowed:false,reason:'Review the pending images before generating another round'};return {allowed:true};}
function assess(report,baseline){
 const findings=report.findings.map(f=>({...f,inherited:baseline.findings.some(b=>b.defect===f.defect)}));
 const add=(severity,defect,measurements)=>findings.push({severity,defect,measurements,inherited:false});
 if(!report.lab?.unchangedSystems.match)add('critical','Non-target scene invariants changed',report.lab?.unchangedSystems??{});
 if(report.lab?.deterministicReproduction!==true)add('critical','Candidate failed deterministic layout reproduction',{});
 const layout=report.lab?.targetLayout;if(layout?.gridAlignedFraction>.4||layout?.duplicatePositionFraction>.1)add('major','Grid-aligned or duplicate vegetation placement',layout);
 if(layout?.count>1000&&layout.occupancyCV<.18)add('major','Excessively uniform spatial occupancy',layout);
 const coverage=report.coverage.fraction,baseCoverage=baseline.coverage.fraction;
 if(report.biome==='savanna'&&coverage!==null&&coverage<Math.max(.08,baseCoverage*.45))add('critical','Severe vegetation coverage loss relative to baseline',{coverage,baseline:baseCoverage,minimumRatio:.45});
 const ratio=report.performance.triangles/Math.max(1,baseline.performance.triangles);
 if(ratio>3||report.performance.calls>baseline.performance.calls+20)add('critical','Extreme rendering-cost regression',{triangleRatio:ratio,calls:report.performance.calls});
 else if(ratio>1.5||report.performance.calls>baseline.performance.calls+5)add('major','Rendering-cost regression needs demonstrated visual benefit',{triangleRatio:ratio,calls:report.performance.calls});
 if(JSON.stringify(report.camera)!==JSON.stringify(baseline.camera)||JSON.stringify(report.renderSettings)!==JSON.stringify(baseline.renderSettings))add('critical','Camera or render settings differ from baseline',{});
 return findings;
}
async function sunlitCheck(png,mask,roi){
 const image=await sharp(png).removeAlpha().raw().toBuffer({resolveWithObject:true}),m=await sharp(mask).removeAlpha().raw().toBuffer();let covered=0,dark=0;
 for(let y=Math.floor(roi[1]*image.info.height);y<roi[3]*image.info.height;y++)for(let x=Math.floor(roi[0]*image.info.width);x<roi[2]*image.info.width;x++){const i=(y*image.info.width+x)*3;if(m[i]>200&&m[i+1]>200&&m[i+2]>200){covered++;if(.2126*image.data[i]+.7152*image.data[i+1]+.0722*image.data[i+2]<25)dark++;}}
 return {roi,maskPixels:covered,nearBlackFraction:covered?dark/covered:null,method:'Actual vegetation silhouette inside a manually reviewed approximate sunlit ROI; not a physical sunlight classification'};
}

function review(id,reviewFile){

function bestFromPreferences(events){const stacks={};for(const e of events){if(e.actor?.type!=='human')continue;const list=stacks[e.experiment]??=[];if(e.decision==='approve')list.push(e);else if(e.decision==='reject'||e.decision==='reject-all')stacks[e.experiment]=list.filter(x=>!(x.record===e.record&&(e.decision==='reject-all'||x.candidateId===e.candidateId)));}return Object.fromEntries(Object.entries(stacks).filter(([,v])=>v.length).map(([key,list])=>{const e=list.at(-1);return [key,{record:e.record,candidateId:e.candidateId,screenshots:e.candidateScreenshots,generatorVersion:e.generatorVersion,parametersHash:hash(e.parameters),parameters:e.parameters,reason:e.reason,approvedBy:e.actor.name,date:e.date}];}));}
