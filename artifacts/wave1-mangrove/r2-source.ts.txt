import * as T from 'three';
import {Reflector} from 'three/addons/objects/Reflector.js';
import type {SceneRecipe} from './recipe';
import type {CameraView} from './render-types';
import {randomSource,hashSeed} from './random';
const smooth=(a:number,b:number,x:number)=>{const t=Math.max(0,Math.min(1,(x-a)/(b-a)));return t*t*(3-2*t);};
function phase(r:SceneRecipe){return hashSeed(r.seed+'|tidal-channels')%6283/1000;}
export function mangroveChannel(r:SceneRecipe,z:number){const p=phase(r);return Math.sin(z*.028+p)*5+Math.sin(z*.069+p*.3)*1.8;}
export function mangroveHeight(r:SceneRecipe,x:number,z:number){const p=phase(r),d=Math.abs(x-mangroveChannel(r,z)),main=1-smooth(5.8,9,d),branch=(1-smooth(3,5.4,Math.abs(x-(-18+(z+28)*.55+Math.sin(z*.048)*2))))*(1-smooth(-12,15,z));return .42-1.02*Math.max(main,branch)+.045*Math.sin(x*.22+z*.13+p)+.025*Math.sin(x*.43-z*.28);}
function compose(recipe:SceneRecipe):SceneRecipe{
 const eligible=recipe.trees.filter(t=>['red-mangrove','avicennia','stilt-mangrove'].includes(t.model.presetId)),r=randomSource(recipe.seed+'|mangrove-shore-layout');
 const trees=eligible.slice(0,28).map((t,i)=>{const z=23-i*3.1+(r()-.5)*5,side=i%2?1:-1,x=mangroveChannel(recipe,z)+side*(9+r()*15);return {...t,position:[x,mangroveHeight(recipe,x,z)-.08,z] as [number,number,number],rotation:r()*Math.PI*2};});
 // Reuse existing eligible individuals as labelled presentation continuation, not extra occurrences.
 const rear=eligible.slice(0,8).map((t,i)=>{const z=-65-r()*65,x=(i/7-.5)*160;return {...t,id:t.id+'-presentation-rear',index:recipe.trees.length+i,position:[x,mangroveHeight(recipe,x,z)-.08,z] as [number,number,number]};});
 return {...recipe,trees:[...trees,...rear],purpose:'visual_laboratory',distribution:'Seeded presentation grouping on synthetic tidal margins using only resolver-eligible regional mangrove candidates; not observed tree locations or abundance.',limitations:[...recipe.limitations,'Waterways, tide, island elevation, root exposure and all specimen positions are art-direction hypotheses; local salinity, tidal range and occurrence are unmeasured.']};
}
function camera(_recipe:SceneRecipe):CameraView{return{position:[0,2.05,31],target:[-2,2,-35]};}
function mount(scene:T.Scene,r:SceneRecipe,sun:T.DirectionalLight,fill:T.DirectionalLight){
 const textures:T.Texture[]=[];for(const o of scene.children)if(o instanceof T.Mesh&&['PlaneGeometry','BoxGeometry'].includes(o.geometry.type))o.visible=false;
 scene.background=new T.Color('#aacac4');scene.fog=new T.Fog('#789d8d',75,340);sun.position.set(-28,38,8);sun.color.set('#ffe4ae');sun.intensity=2.7;sun.shadow.intensity=.65;fill.color.set('#6fa3a6');fill.intensity=.5;scene.children.forEach(o=>{if(o instanceof T.HemisphereLight){o.color.set('#c2d6ca');o.groundColor.set('#465b4b');o.intensity=.75;}});
 let pending=2;scene.userData.labAssetsReady=false;const loaded=()=>{pending--;scene.userData.labAssetsReady=pending===0;scene.userData.recipeShadowDirty=true;scene.userData.artRevision=(scene.userData.artRevision??0)+1;};const skyTexture=new T.TextureLoader().load('/assets/puget-v1/sky-matte-v1.webp',loaded,undefined,()=>{loaded();scene.userData.wave1AssetError='sky load failed';});skyTexture.colorSpace=T.SRGBColorSpace;textures.push(skyTexture);
 const sky=new T.Mesh(new T.SphereGeometry(850,32,16),new T.MeshBasicMaterial({map:skyTexture,side:T.BackSide,depthWrite:false,fog:false,color:'#b9d1bf',toneMapped:false}));sky.name='mangrove-existing-sky';sky.renderOrder=-10;scene.add(sky);
 const mud=new T.TextureLoader().load('/assets/puget-v1/forest-floor.png',loaded,undefined,()=>{loaded();scene.userData.wave1AssetError='substrate detail failed';});mud.colorSpace=T.SRGBColorSpace;mud.wrapS=mud.wrapT=T.RepeatWrapping;mud.anisotropy=4;textures.push(mud);
 const mudMaterial=new T.MeshStandardMaterial({map:mud,vertexColors:true,roughness:.94});mudMaterial.onBeforeCompile=shader=>{shader.fragmentShader=shader.fragmentShader.replace('#include <map_fragment>','#ifdef USE_MAP\nvec4 soil=texture2D(map,vMapUv);diffuseColor.rgb*=clamp(dot(soil.rgb,vec3(.2126,.7152,.0722))/.2,.75,1.25);\n#endif');};
 for(const [width,depth,zc,segments]of[[220,260,-75,180],[620,740,-235,100]]){const g=new T.PlaneGeometry(width,depth,segments,segments);g.rotateX(-Math.PI/2);g.translate(0,0,zc);const pos=g.getAttribute('position'),uv=g.getAttribute('uv'),colors=new Float32Array(pos.count*3);for(let i=0;i<pos.count;i++){const x=pos.getX(i),z=pos.getZ(i),y=mangroveHeight(r,x,z),wet=1-smooth(.16,.4,y),c=new T.Color('#494934').lerp(new T.Color('#263b31'),wet*.75);c.lerp(new T.Color('#776f46'),(.5+.5*Math.sin(x*.041+z*.025))*.12);pos.setY(i,y);uv.setXY(i,x/7,z/7);c.toArray(colors,i*3);}if(width>220){const ids:number[]=[],old=g.index!;for(let i=0;i<old.count;i+=3){const a=[old.getX(i),old.getX(i+1),old.getX(i+2)];if(a.every(j=>Math.abs(pos.getX(j))<108&&Math.abs(pos.getZ(j)+75)<128))continue;ids.push(...a);}g.setIndex(ids);}g.setAttribute('color',new T.BufferAttribute(colors,3));g.computeVertexNormals();const ground=new T.Mesh(g,mudMaterial);ground.name='mangrove-tidal-substrate';ground.receiveShadow=true;scene.add(ground);}

 const water=new Reflector(new T.PlaneGeometry(620,740),{textureWidth:512,textureHeight:256,color:0x79958a,clipBias:.003});water.rotation.x=-Math.PI/2;water.position.set(0,.29,-235);water.name='mangrove-reflected-tidal-water';const wm=water.material as T.ShaderMaterial;wm.fragmentShader=wm.fragmentShader.replace('texture2DProj( tDiffuse, vUv )','texture2DProj( tDiffuse, vUv + vec4(sin(vUv.y*170.)*.00055,cos(vUv.x*230.)*.0004,0.,0.) )');wm.fragmentShader=wm.fragmentShader.replace('#include <tonemapping_fragment>','gl_FragColor.rgb=mix(vec3(.025,.06,.045),clamp(gl_FragColor.rgb,0.,.35),.22);\n#include <tonemapping_fragment>');scene.add(water);
 const rand=randomSource(r.seed+'|shore-dressing'),geo=new T.ConeGeometry(.07,.65,5),mat=new T.MeshStandardMaterial({color:'#64634b',roughness:1}),positions:{x:number;z:number;h:number}[]=[];
 // Unnamed driftwood/root-like substrate accents, not unsupported pneumatophore species claims.
 for(let i=0;i<700;i++){const z=28-rand()*125,x=(rand()-.5)*90,y=mangroveHeight(r,x,z);if(y>.32&&y<.52&&rand()<.65)positions.push({x,z,h:.12+rand()*.25});}
 const shoots=new T.InstancedMesh(geo,mat,positions.length),o=new T.Object3D();shoots.name='mangrove-shore-root-debris';positions.forEach((p,i)=>{o.position.set(p.x,mangroveHeight(r,p.x,p.z)+p.h*.1,p.z);o.rotation.set((rand()-.5)*.8,rand()*6.283,(rand()-.5)*.7);o.scale.set(.45+rand()*.5,p.h/.65,.45+rand()*.5);o.updateMatrix();shoots.setMatrixAt(i,o.matrix);});shoots.computeBoundingSphere();scene.add(shoots);
 // Quiet distant forest masses join the real rear trees, leaving the water opening readable.
 const massGeo=new T.SphereGeometry(1,12,7),massMat=new T.MeshStandardMaterial({color:'#456a53',roughness:1}),forest=new T.InstancedMesh(massGeo,massMat,46);forest.name='mangrove-distant-canopy-masses';for(let i=0;i<46;i++){const x=(i/45-.5)*420,z=-150-rand()*65;o.position.set(x,8+rand()*4,z);o.scale.set(10+rand()*12,3+rand()*3,8+rand()*6);o.rotation.set(0,rand()*6.283,0);o.updateMatrix();forest.setMatrixAt(i,o.matrix);}forest.computeBoundingSphere();scene.add(forest);
 scene.userData.wave1={id:'mangrove',biome:14,version:'mangrove-1',trees:r.trees.length,eligibleSpecies:[...new Set(r.trees.map(t=>t.species.scientificName))],shoreInstances:positions.length,farCanopyInstances:46,layers:{season:'existing unresolved ecological input; no forced flowering',weather:'clear reference study, no weather simulation',timeOfDay:{sun:[-28,38,8],color:'#ffe4ae',intensity:2.7},tide:{waterY:.29,source:'synthetic art state, no tidal observation'}},reflection:{type:'planar render-target mirror',resolution:[512,256],limitations:'One extra scene render; mirror softened only by tint/ripple, no depth-aware refraction.'},evidence:'Regionally native candidates only; local occurrence unverified',terrain:'Synthetic tidal field; not measured hydrology'};
 scene.userData.recipeShadowDirty=true;return()=>{textures.forEach(t=>t.dispose());water.getRenderTarget().dispose();};
}
export const mangroveStudy={id:'mangrove',biome:14,compose,camera,mount};
