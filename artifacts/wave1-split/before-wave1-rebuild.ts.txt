function rootIntrudes(t:SceneTree,position:[number,number,number],rotation:number,plan:ReturnType<typeof fieldPlan>){
 const c=Math.cos(rotation),s=Math.sin(rotation);
 for(const segment of t.model.segments){if(Math.min(segment.a[1],segment.b[1])>1.9)continue;
  for(let i=0;i<=6;i++){const f=i/6,x=segment.a[0]+(segment.b[0]-segment.a[0])*f,z=segment.a[2]+(segment.b[2]-segment.a[2])*f;
   if(collisionFootprint(plan,position[0]+x*c-z*s,position[2]+x*s+z*c,Math.max(segment.r0,segment.r1)+.12))return true;}}
 return false;
}

function compose(recipe:SceneRecipe,biome:FieldBiome):SceneRecipe{
 const plan=fieldPlan(biome,recipe.seed),r=randomSource(recipe.seed+'|field-canopy'),trees:SceneTree[]=[];
 const eligible=recipe.trees.filter(t=>biome==='mangrove'?['red-mangrove','avicennia','stilt-mangrove'].includes(t.model.presetId):true);
 if(biome!=='desert'&&eligible.length){for(let attempt=0;attempt<16000&&trees.length<(biome==='mangrove'?95:24);attempt++){
  const near=trees.length<(biome==='mangrove'?38:5),x=plan.eye[0]+(r()-.5)*(near?85:650),z=plan.eye[2]+(r()-.5)*(near?85:650),f=landscapeField(biome,recipe.seed,x,z);
  const q=cameraSpace(plan,x,z);if(q.forward>0&&q.forward<35&&Math.abs(q.side)<7+q.forward*.12)continue;
  if(characterSpace(plan,x,z)||Math.hypot(x-plan.eye[0],z-plan.eye[2])<11||f.woody<.45||f.height<.03)continue;
  if(trees.some(t=>Math.hypot(t.position[0]-x,t.position[2]-z)<(near?5.2:14)))continue;
  const t=eligible[trees.length%eligible.length],rotation=r()*Math.PI*2,position:[number,number,number]=[x,f.height-.05,z];if(rootIntrudes(t,position,rotation,plan))continue;
  const studio=!near&&t.studio?{...t.studio,fronds:t.studio.fronds.filter((_,i)=>i%12===0).map(f=>({...f,width:f.width*3,length:f.length*2.5}))}:biome==='mangrove'&&t.studio?{...t.studio,fronds:t.studio.fronds.map(f=>({...f,width:f.width*1.1,length:f.length*1.06}))}:t.studio;trees.push({...t,studio,model:near?t.model:{...t.model,segments:t.model.segments.filter(s=>s.depth<=1)},id:t.id+'-field-'+trees.length,position,rotation});
 }}
 return {...recipe,trees:biome==='desert'?recipe.trees:trees,purpose:'visual_laboratory',distribution:'Connected synthetic habitat field. Reused source individuals are presentation occurrences, not population observations.',limitations:[...recipe.limitations,'Field heights/moisture and camera selection are artistic synthetic inputs, not measured terrain or local habitat. Conditional unnamed wetland canopy is not verified botanical data.']};
}
function fracturedBlock(variant:number){
 const g=new T.BoxGeometry(2,2,2,10,10,10),p=g.getAttribute('position');
 for(let i=0;i<p.count;i++){
  let x=p.getX(i),y=p.getY(i),z=p.getZ(i);const v=new T.Vector3(x,y,z),core=new T.Vector3(T.MathUtils.clamp(x,-.67,.67),T.MathUtils.clamp(y,-.67,.67),T.MathUtils.clamp(z,-.67,.67));v.sub(core).normalize().multiplyScalar(.33).add(core);x=v.x;y=v.y;z=v.z;
  // Unequal joint planes and worn edges, rather than triangular spherical facets.
  const cut=Math.max(0,x*.55+y*.8+z*.23-.79);x-=cut*.55;y-=cut*.8;z-=cut*.23;
  const cut2=Math.max(0,-x*.85+y*.18-z*.35-.86);x+=cut2*.85;y-=cut2*.18;z+=cut2*.35;
  const wear=.025*Math.sin(x*15+variant)*Math.sin(z*12+y*9);p.setXYZ(i,x+wear,y+wear*.4,z+wear);
 }g.computeVertexNormals();return weatherRockGeometry(g,variant);
}
function mount(scene:T.Scene,recipe:SceneRecipe,biome:FieldBiome,sun:T.DirectionalLight,fill:T.DirectionalLight){
 const plan=fieldPlan(biome,recipe.seed),field=(x:number,z:number)=>landscapeField(biome,recipe.seed,x,z),rand=randomSource(recipe.seed+'|field-dressing'),o=new T.Object3D(),textures:T.Texture[]=[];
 for(const child of scene.children)if(child instanceof T.Mesh&&['PlaneGeometry','BoxGeometry'].includes(child.geometry.type))child.visible=false;
 scene.userData.labAssetsReady=false;let pendingAssets=biome==='mangrove'?2:1;const assetLoaded=()=>{scene.userData.labAssetsReady=--pendingAssets===0;scene.userData.artRevision=(scene.userData.artRevision??0)+1;};const skyMap=new T.TextureLoader().load('/assets/puget-v1/sky-matte-v1.webp',assetLoaded,undefined,()=>{assetLoaded();scene.userData.wave1AssetError='sky unavailable';});skyMap.colorSpace=T.SRGBColorSpace;textures.push(skyMap);
 const skyMaterial=new T.MeshBasicMaterial({map:skyMap,side:T.BackSide,depthWrite:false,fog:false,toneMapped:false});skyMaterial.onBeforeCompile=shader=>{shader.fragmentShader=shader.fragmentShader.replace('#include <map_fragment>',`#ifdef USE_MAP
 vec4 skySample=texture2D(map,vMapUv),oppositeEdge=texture2D(map,vec2(1.-vMapUv.x,vMapUv.y));
 float edge=1.-smoothstep(0.,.055,min(vMapUv.x,1.-vMapUv.x));
 diffuseColor*=mix(skySample,(skySample+oppositeEdge)*.5,edge);
 #endif`);};skyMaterial.customProgramCacheKey=()=> 'field-sky-seam-feather-1';
 const sky=new T.Mesh(new T.SphereGeometry(2000,40,20),skyMaterial);sky.rotation.y=Math.PI/2-plan.heading;sky.name='field-existing-sky';scene.add(sky);
 let mudDetail:T.Texture|undefined;if(biome==='mangrove'){mudDetail=new T.TextureLoader().load('/assets/wave1-detail/tidal-mud-v1.webp',assetLoaded,undefined,()=>{assetLoaded();scene.userData.wave1AssetError='tidal mud unavailable';});mudDetail.colorSpace=T.SRGBColorSpace;mudDetail.wrapS=mudDetail.wrapT=T.RepeatWrapping;mudDetail.anisotropy=4;textures.push(mudDetail);}
 const pigment=(x:number,z:number)=>{const f=field(x,z);if(biome==='desert')return new T.Color('#d29a67').lerp(new T.Color('#a46f4c'),f.rock*.55).lerp(new T.Color('#e2be8f'),f.moisture*.25);
 const soil=new T.Color(biome==='mangrove'?'#8a7458':'#8b7656').lerp(new T.Color('#514d42'),f.moisture*.55);
 return biome==='mangrove'?soil:soil.lerp(new T.Color('#839b52').lerp(new T.Color('#c7b85e'),f.community*.25),smooth(.0,.12,f.height)*(.75+.25*f.community)*(1.-.25*f.substrate));};
 const mat=biome==='desert'?new T.MeshStandardMaterial({vertexColors:true,roughness:1}):wetGroundMaterial(mudDetail);
 // Same continuous field near/far; aligned boundary vertices avoid overlapping independent terrain sheets.
 const cx=Math.round(plan.eye[0]/10)*10,cz=Math.round(plan.eye[2]/10)*10;
 for(const [size,steps,hole]of [[96,192,0],[240,120,48],[1800,180,120]]as const){const g=new T.PlaneGeometry(size,size,steps,steps);g.rotateX(-Math.PI/2);g.translate(cx,0,cz);const p=g.getAttribute('position'),colors=new Float32Array(p.count*3);for(let i=0;i<p.count;i++){const x=p.getX(i),z=p.getZ(i);p.setY(i,field(x,z).height);pigment(x,z).toArray(colors,i*3);}if(hole){const ix:number[]=[],idx=g.index!;for(let i=0;i<idx.count;i+=3){const a=idx.getX(i),b=idx.getX(i+1),c=idx.getX(i+2),x=(p.getX(a)+p.getX(b)+p.getX(c))/3,z=(p.getZ(a)+p.getZ(b)+p.getZ(c))/3;if(Math.abs(x-cx)<hole&&Math.abs(z-cz)<hole)continue;ix.push(a,b,c);}g.setIndex(ix);}g.setAttribute('color',new T.BufferAttribute(colors,3));g.computeVertexNormals();const land=new T.Mesh(g,mat);land.name='connected-field-terrain';land.receiveShadow=true;scene.add(land);}
 const disposeImpostors=biome!=='desert'?forestImpostors(scene,recipe,biome):()=>{};
 let optics:ReturnType<typeof tidalWater>|undefined;
 if(biome!=='desert'){optics=tidalWater((x,z)=>field(x,z).height,0,2200,0,recipe.wind?1:0,[plan.eye[0],plan.eye[2]]);scene.add(optics.water);}
 const counts:{name:string;count:number}[]=[];
 if(biome==='desert'){
  for(let variant=0;variant<3;variant++){const placements:{x:number;z:number;s:number}[]=[];for(let i=0;i<12000&&placements.length<75;i++){const x=plan.eye[0]+(rand()-.5)*260,z=plan.eye[2]+(rand()-.5)*260,f=field(x,z),q=cameraSpace(plan,x,z);if((q.forward>0&&q.forward<50&&Math.abs(q.side)<3+q.forward*.28)||f.rock<.3||characterSpace(plan,x,z)||Math.hypot(q.side,q.forward)<8||rand()>f.rock*.6)continue;placements.push({x,z,s:(.25+rand()*2.8)*(q.forward>80?.7:1)*smooth(4,22,Math.hypot(q.side,q.forward))});}
   const rockMat=desertRockMaterial(),mesh=new T.InstancedMesh(fracturedBlock(variant),rockMat,placements.length);mesh.name='field-jointed-bedrock';placements.forEach((p,i)=>{o.position.set(p.x,field(p.x,p.z).height+p.s*.28,p.z);o.rotation.set(rand()*.3,rand()*1.5,rand()*.35);o.scale.set(p.s*(1+rand()),p.s*.8,p.s);o.updateMatrix();mesh.setMatrixAt(i,o.matrix);mesh.setColorAt(i,new T.Color('#a96f4b').lerp(new T.Color('#dab184'),rand()*.8));});mesh.computeBoundingSphere();mesh.castShadow=mesh.receiveShadow=true;scene.add(mesh);counts.push({name:mesh.name,count:mesh.count});}
 }
 if(biome==='flooded'){
 const plants:{x:number;z:number;size:number;far:boolean}[]=[];
 for(let i=0;i<65000&&plants.length<12000;i++){
  const near=rand()<.78,x=plan.eye[0]+(rand()-.5)*(near?140:700),z=plan.eye[2]+(rand()-.5)*(near?140:700),f=field(x,z);if(f.height<-.015)continue;
  const suitability=(.09+.82*f.community)*(.32+.68*(1-smooth(.27,.52,f.height)));if(rand()>suitability)continue;
  let size=(.2+f.community*.9+rand()*.18)*stagingHeightFactor(plan,x,z);if(collisionFootprint(plan,x,z,.15))size=Math.min(size,.10);
  plants.push({x,z,size,far:Math.hypot(x-plan.eye[0],z-plan.eye[2])>40});
 }
 // A second, short/dry community fills suitable islands without erasing low-coverage soil patches.
 for(let i=0,added=0;i<9000&&added<2200;i++){const forward=rand()*23,side=(rand()-.5)*(7+forward*.7),x=plan.eye[0]+Math.sin(plan.heading)*forward+Math.cos(plan.heading)*side,z=plan.eye[2]-Math.cos(plan.heading)*forward+Math.sin(plan.heading)*side,f=field(x,z);if(f.height<.015||rand()>.3+.5*(1-f.community))continue;let size=(.12+rand()*.2)*stagingHeightFactor(plan,x,z);if(collisionFootprint(plan,x,z,.15))size=Math.min(.1,size);plants.push({x,z,size,far:true});added++;}
 // Lateral grass swaths have a distinct framing role, with field-filtered irregular anchors.
 for(let cluster=0;cluster<9;cluster++){const forward=3+rand()*17,side=(cluster%2?1:-1)*Math.max(2.6,forward*(.48+rand()*.24)),cx=plan.eye[0]+Math.sin(plan.heading)*forward+Math.cos(plan.heading)*side,cz=plan.eye[2]-Math.cos(plan.heading)*forward+Math.sin(plan.heading)*side;
 for(let i=0;i<40;i++){const x=cx+(rand()-.5)*2.4,z=cz+(rand()-.5)*2.4,f=field(x,z);if(f.height<.015||collisionFootprint(plan,x,z,1))continue;plants.push({x,z,size:.95+rand()*.7,far:false});}}
 for(const far of [false,true]){const selected=plants.filter(p=>p.far===far),grass=new T.InstancedMesh(meadowGeometry(far),meadowMaterial(far),selected.length);grass.name=far?'field-distant-meadow':'field-dimensional-meadow';selected.forEach((p,i)=>{o.position.set(p.x,field(p.x,p.z).height-.02,p.z);o.rotation.set(0,.35*Math.sin(p.x*.12+p.z*.09)+(rand()-.5)*1.7,0);o.scale.setScalar(p.size);o.updateMatrix();grass.setMatrixAt(i,o.matrix);const f=field(p.x,p.z);grass.setColorAt(i,new T.Color('#718e4d').lerp(new T.Color('#c7b85e'),f.community*.35).lerp(pigment(p.x,p.z),.2));});grass.computeBoundingSphere();grass.receiveShadow=true;grass.castShadow=false;scene.add(grass);counts.push({name:grass.name,count:grass.count});}
 }
 const disposeDetail=biome==='desert'?installDesertDetail(scene,recipe,plan):biome==='mangrove'?installMangroveDetail(scene,recipe,plan):installMeadowDetail(scene,recipe,plan);
 const s=Math.sin(plan.heading),c=Math.cos(plan.heading);sun.position.set(plan.eye[0]-c*50+s*20,50,plan.eye[2]-s*50-c*20);sun.color.set('#ffd38a');sun.intensity=4;sun.shadow.intensity=.65;sun.target.position.set(plan.eye[0]+s*12,0,plan.eye[2]-c*12);scene.add(sun.target);fill.intensity=.3;fill.color.set('#8fafc2');scene.children.forEach(child=>{if(child instanceof T.HemisphereLight){child.intensity=.65;child.color.set('#a4c7dd');child.groundColor.set('#998160');}});scene.fog=new T.Fog(biome==='desert'?'#b4bbc0':'#91a7a0',180,biome==='mangrove'?650:1200);
 scene.userData.wave1={id:biome,biome:recipe.ecoregion?.BIOME_NUM,version:'connected-field-detail-1',presentation:{foliageAtmosphere:{near:120,far:650,strength:.5,color:biome==='mangrove'?'#72978b':'#78977c'}},plan,trees:recipe.trees.length,instances:counts,ecology:'Named source candidates retained; any unnamed wooded margin is a conditional growth-form study, never point occurrence.',layers:{season:'unresolved',weather:'clear study',timeOfDay:'warm directional study'},water:optics?.limitations,depthSampling:optics?.depthSampling};scene.userData.recipeShadowDirty=true;
 let minGround=Infinity;for(let along=2.8;along<=6.2;along+=.2)for(let side=-.85;side<=.85;side+=.17){const x=plan.eye[0]+s*along+c*side,z=plan.eye[2]-c*along+s*side;minGround=Math.min(minGround,field(x,z).height);}
 scene.userData.stagingAudit={capsuleRadius:.35,bodyHeight:1.7,approachRadius:.85,minimumGroundAboveWater:minGround,rootIntrusions:recipe.trees.filter(t=>rootIntrudes(t,t.position,t.rotation,plan)).length,shortGrassHeightBound:.08,note:'Conservative source-segment sweep and dry approach samples, not a character physics simulation.'};
 const stagePoint=new T.Vector3(plan.eye[0]+s*4.5,0,plan.eye[2]-c*4.5);stagePoint.y=field(stagePoint.x,stagePoint.z).height+.85;
 const proxy=new T.Mesh(new T.CapsuleGeometry(.35,1,5,10),new T.MeshBasicMaterial({color:'#d74cbd',wireframe:true}));proxy.position.copy(stagePoint);proxy.name='diagnostic-character-capsule';proxy.visible=new URLSearchParams(location.search).get('stagingDebug')==='1';scene.add(proxy);
 scene.userData.showStagingDebug=(visible:boolean)=>{proxy.visible=visible;};
 return()=>{disposeDetail();disposeImpostors();textures.forEach(t=>t.dispose());optics?.dispose();};
}
export function rebuiltStudy(biome:FieldBiome,number:number){return{id:biome,biome:number,compose:(r:SceneRecipe)=>compose(r,biome),camera:(r:SceneRecipe)=>{const p=fieldPlan(biome,r.seed);return{position:p.eye,target:p.target};},mount:(s:T.Scene,r:SceneRecipe,sun:T.DirectionalLight,fill:T.DirectionalLight)=>mount(s,r,biome,sun,fill)};}
