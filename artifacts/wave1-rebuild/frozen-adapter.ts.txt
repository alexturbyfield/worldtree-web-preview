import {forestImpostors} from './wave1-impostors';
import * as T from 'three';
import type {SceneRecipe,SceneTree} from './recipe';
import {landscapeField,fieldPlan,cameraSpace,characterSpace,smooth,type FieldBiome} from './wave1-fields';
import {randomSource} from './random';
import {tidalWater} from './wave1-water';
import {sedgeGeometry} from './wave1-flooded';

function compose(recipe:SceneRecipe,biome:FieldBiome):SceneRecipe{
 const plan=fieldPlan(biome,recipe.seed),r=randomSource(recipe.seed+'|field-canopy'),trees:SceneTree[]=[];
 const eligible=recipe.trees.filter(t=>biome==='mangrove'?['red-mangrove','avicennia','stilt-mangrove'].includes(t.model.presetId):true);
 if(biome!=='desert'&&eligible.length){for(let attempt=0;attempt<16000&&trees.length<(biome==='mangrove'?95:24);attempt++){
  const near=trees.length<(biome==='mangrove'?38:5),x=plan.eye[0]+(r()-.5)*(near?85:650),z=plan.eye[2]+(r()-.5)*(near?85:650),f=landscapeField(biome,recipe.seed,x,z);
  const q=cameraSpace(plan,x,z);if(q.forward>0&&q.forward<35&&Math.abs(q.side)<7+q.forward*.12)continue;
  if(characterSpace(plan,x,z)||Math.hypot(x-plan.eye[0],z-plan.eye[2])<11||f.woody<.45||f.height<.03)continue;
  if(trees.some(t=>Math.hypot(t.position[0]-x,t.position[2]-z)<(near?5.2:14)))continue;
  const t=eligible[trees.length%eligible.length],studio=!near&&t.studio?{...t.studio,fronds:t.studio.fronds.filter((_,i)=>i%12===0).map(f=>({...f,width:f.width*3,length:f.length*2.5}))}:t.studio;trees.push({...t,studio,model:near?t.model:{...t.model,segments:t.model.segments.filter(s=>s.depth<=1)},id:t.id+'-field-'+trees.length,position:[x,f.height-.05,z],rotation:r()*Math.PI*2});
 }}
 return {...recipe,trees:biome==='desert'?recipe.trees:trees,purpose:'visual_laboratory',distribution:'Connected synthetic habitat field. Reused source individuals are presentation occurrences, not population observations.',limitations:[...recipe.limitations,'Field heights/moisture and camera selection are artistic synthetic inputs, not measured terrain or local habitat. Conditional unnamed wetland canopy is not verified botanical data.']};
}
function fracturedBlock(variant:number){
 const g=new T.BoxGeometry(2,2,2,10,10,10),p=g.getAttribute('position');
 for(let i=0;i<p.count;i++){
  let x=p.getX(i),y=p.getY(i),z=p.getZ(i);const v=new T.Vector3(x,y,z),core=new T.Vector3(T.MathUtils.clamp(x,-.67,.67),T.MathUtils.clamp(y,-.67,.67),T.MathUtils.clamp(z,-.67,.67));v.sub(core).normalize().multiplyScalar(.33).add(core);x=v.x;y=v.y;z=v.z;
  // Unequal joint planes and worn edges, rather than triangular spherical facets.
  const cut=Math.max(0,x*.55+y*.8+z*.23-.79);x-=cut*.55;y-=cut*.8;z-=cut*.23;
  const cut2=Math.max(0,-x*.85+y*.18-z*.35-.86);x+=cut2*.85;y-=cut2*.18;z+=cut2*.35;
  const wear=.025*Math.sin(x*15+variant)*Math.sin(z*12+y*9);p.setXYZ(i,x+wear,y+wear*.4,z+wear);
 }g.computeVertexNormals();return g;
}
function mount(scene:T.Scene,recipe:SceneRecipe,biome:FieldBiome,sun:T.DirectionalLight,fill:T.DirectionalLight){
 const plan=fieldPlan(biome,recipe.seed),field=(x:number,z:number)=>landscapeField(biome,recipe.seed,x,z),rand=randomSource(recipe.seed+'|field-dressing'),o=new T.Object3D(),textures:T.Texture[]=[];
 for(const child of scene.children)if(child instanceof T.Mesh&&['PlaneGeometry','BoxGeometry'].includes(child.geometry.type))child.visible=false;
 scene.userData.labAssetsReady=false;const skyMap=new T.TextureLoader().load('/assets/puget-v1/sky-matte-v1.webp',()=>{scene.userData.labAssetsReady=true;scene.userData.artRevision=(scene.userData.artRevision??0)+1;},undefined,()=>{scene.userData.labAssetsReady=true;scene.userData.wave1AssetError='sky unavailable';});skyMap.colorSpace=T.SRGBColorSpace;textures.push(skyMap);
 const sky=new T.Mesh(new T.SphereGeometry(2000,40,20),new T.MeshBasicMaterial({map:skyMap,side:T.BackSide,depthWrite:false,fog:false,toneMapped:false}));sky.name='field-existing-sky';scene.add(sky);
 const pigment=(x:number,z:number)=>{const f=field(x,z);return biome==='desert'?new T.Color('#d29a67').lerp(new T.Color('#a46f4c'),f.rock*.55).lerp(new T.Color('#e2be8f'),f.moisture*.25):new T.Color(biome==='mangrove'?'#967d59':'#b9b26b').lerp(new T.Color('#655640'),f.moisture*.7).lerp(new T.Color('#778747'),biome==='flooded'?f.community*(1-f.moisture)*.65:0);};
 const mat=new T.MeshStandardMaterial({vertexColors:true,roughness:1});
 // Same continuous field near/far; aligned boundary vertices avoid overlapping independent terrain sheets.
 const cx=Math.round(plan.eye[0]/10)*10,cz=Math.round(plan.eye[2]/10)*10;
 for(const [size,steps,hole]of [[240,240,false],[1800,180,true]]as const){const g=new T.PlaneGeometry(size,size,steps,steps);g.rotateX(-Math.PI/2);g.translate(cx,0,cz);const p=g.getAttribute('position'),colors=new Float32Array(p.count*3);for(let i=0;i<p.count;i++){const x=p.getX(i),z=p.getZ(i);p.setY(i,field(x,z).height);pigment(x,z).toArray(colors,i*3);}if(hole){const ix:number[]=[],idx=g.index!;for(let i=0;i<idx.count;i+=3){const a=idx.getX(i),b=idx.getX(i+1),c=idx.getX(i+2),x=(p.getX(a)+p.getX(b)+p.getX(c))/3,z=(p.getZ(a)+p.getZ(b)+p.getZ(c))/3;if(Math.abs(x-cx)<120&&Math.abs(z-cz)<120)continue;ix.push(a,b,c);}g.setIndex(ix);}g.setAttribute('color',new T.BufferAttribute(colors,3));g.computeVertexNormals();const land=new T.Mesh(g,mat);land.name='connected-field-terrain';land.receiveShadow=true;scene.add(land);}
 const disposeImpostors=biome!=='desert'?forestImpostors(scene,recipe,biome):()=>{};
 let optics:ReturnType<typeof tidalWater>|undefined;
 if(biome!=='desert'){optics=tidalWater((x,z)=>field(x,z).height,0,2200,0,recipe.wind?1:0);scene.add(optics.water);}
 const counts:{name:string;count:number}[]=[];
 if(biome==='desert'){
  for(let variant=0;variant<3;variant++){const placements:{x:number;z:number;s:number}[]=[];for(let i=0;i<12000&&placements.length<75;i++){const x=plan.eye[0]+(rand()-.5)*260,z=plan.eye[2]+(rand()-.5)*260,f=field(x,z),q=cameraSpace(plan,x,z);if((q.forward>0&&q.forward<50&&Math.abs(q.side)<3+q.forward*.28)||f.rock<.3||characterSpace(plan,x,z)||Math.hypot(q.side,q.forward)<8||rand()>f.rock*.6)continue;placements.push({x,z,s:(.25+rand()*2.8)*(q.forward>80?.7:1)*smooth(4,22,Math.hypot(q.side,q.forward))});}
   const rockMat=new T.MeshStandardMaterial({roughness:1}),mesh=new T.InstancedMesh(fracturedBlock(variant),rockMat,placements.length);mesh.name='field-jointed-bedrock';placements.forEach((p,i)=>{o.position.set(p.x,field(p.x,p.z).height+p.s*.28,p.z);o.rotation.set(rand()*.3,rand()*1.5,rand()*.35);o.scale.set(p.s*(1+rand()),p.s*.8,p.s);o.updateMatrix();mesh.setMatrixAt(i,o.matrix);mesh.setColorAt(i,new T.Color('#a96f4b').lerp(new T.Color('#dab184'),rand()*.8));});mesh.computeBoundingSphere();mesh.castShadow=mesh.receiveShadow=true;scene.add(mesh);counts.push({name:mesh.name,count:mesh.count});}
 }
 const plants:{x:number;z:number;size:number}[]=[];
 for(let i=0;i<50000&&plants.length<(biome==='desert'?180:biome==='mangrove'?2200:10500);i++){
  const near=rand()<.72,x=plan.eye[0]+(rand()-.5)*(near?140:700),z=plan.eye[2]+(rand()-.5)*(near?140:700),f=field(x,z);if(f.height<(biome==='desert'?0:.015)||characterSpace(plan,x,z))continue;
  const suitability=biome==='desert'?f.moisture*f.community*.15:biome==='mangrove'?f.moisture*f.community*.55:f.community*(.25+.75*(1-smooth(.25,.5,f.height)));
  if(rand()>suitability)continue;plants.push({x,z,size:(biome==='desert'?.25:.45)+f.community*(biome==='mangrove'?.65:.85)+rand()*.3});
 }
 const grass=new T.InstancedMesh(sedgeGeometry(false),new T.MeshStandardMaterial({side:T.DoubleSide,roughness:1}),plants.length);grass.name='field-correlated-low-vegetation';plants.forEach((p,i)=>{o.position.set(p.x,field(p.x,p.z).height-.025,p.z);o.rotation.set(0,rand()*6.283,0);o.scale.setScalar(p.size);o.updateMatrix();grass.setMatrixAt(i,o.matrix);grass.setColorAt(i,new T.Color(biome==='desert'?'#a39465':biome==='mangrove'?'#678548':'#b4ad58').lerp(pigment(p.x,p.z),.22));});grass.computeBoundingSphere();grass.receiveShadow=true;scene.add(grass);counts.push({name:grass.name,count:grass.count});
 const s=Math.sin(plan.heading),c=Math.cos(plan.heading);sun.position.set(plan.eye[0]-c*50+s*20,50,plan.eye[2]-s*50-c*20);sun.color.set('#ffd38a');sun.intensity=4;sun.shadow.intensity=.65;fill.intensity=.3;fill.color.set('#8fafc2');scene.children.forEach(child=>{if(child instanceof T.HemisphereLight){child.intensity=.65;child.color.set('#a4c7dd');child.groundColor.set('#998160');}});scene.fog=new T.Fog(biome==='desert'?'#b4bbc0':'#91a7a0',180,biome==='mangrove'?650:1200);
 scene.userData.wave1={id:biome,biome:recipe.ecoregion?.BIOME_NUM,version:'connected-field-structure-1',plan,trees:recipe.trees.length,instances:counts,ecology:'Named source candidates retained; any unnamed wooded margin is a conditional growth-form study, never point occurrence.',layers:{season:'unresolved',weather:'clear study',timeOfDay:'warm directional study'},water:optics?.limitations};scene.userData.recipeShadowDirty=true;
 return()=>{disposeImpostors();textures.forEach(t=>t.dispose());optics?.dispose();};
}
export function rebuiltStudy(biome:FieldBiome,number:number){return{id:biome,biome:number,compose:(r:SceneRecipe)=>compose(r,biome),camera:(r:SceneRecipe)=>{const p=fieldPlan(biome,r.seed);return{position:p.eye,target:p.target};},mount:(s:T.Scene,r:SceneRecipe,sun:T.DirectionalLight,fill:T.DirectionalLight)=>mount(s,r,biome,sun,fill)};}
