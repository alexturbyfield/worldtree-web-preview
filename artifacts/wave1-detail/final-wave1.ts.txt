import {desertStudy} from './wave1-desert';
import {mangroveStudy} from './wave1-mangrove';
import {floodedStudy} from './wave1-flooded';
import type {SceneRecipe} from './recipe';
import {rebuiltStudy} from './wave1-rebuild';
/** Opt-in presentation only; RESOLVE gate cannot be bypassed by a query string. */
export function wave1Study(recipe:SceneRecipe,search:string){const params=new URLSearchParams(search),id=params.get('wave1'),original=[desertStudy,mangroveStudy,floodedStudy].find(s=>s.id===id&&recipe.ecoregion?.BIOME_NUM===s.biome);return original&&params.get('fieldStudy')==='rebuild'?rebuiltStudy(original.id as 'desert'|'mangrove'|'flooded',original.biome):original;}
