const clamp=(v:number)=>Math.max(0,Math.min(1,v));
function smooth(a:number,b:number,v:number){const t=clamp((v-a)/(b-a));return t*t*(3-2*t);}
export function floodedFields(seed:string,x:number,z:number){
 const phase=(hashSeed(seed+'|wetland-field')%6283)/1000,channel=Math.sin(z*.018+phase)*12+Math.sin(z*.043+phase*.7)*4;
 const width=3.2+1.4*Math.sin(z*.025+phase),bank=smooth(width,width+7,Math.abs(x-channel)),patch=.5+.24*Math.sin(x*.046+Math.sin(z*.021)*2+phase)+.18*Math.sin(z*.055-x*.017+phase*.4);
 const sideChannel=5*Math.exp(-Math.pow((z+60+Math.sin(x*.032+phase)*18)/10,2));
 const pool=.5+.26*Math.sin(x*.031+z*.017+Math.sin(z*.027))+ .20*Math.sin(x*.052-z*.021+phase);
 const y=-.18+bank*.55+(patch-.5)*.29-sideChannel*.06-smooth(.56,.77,pool)*.32;
 return{y,patch,bank,channel,shore:1-smooth(.015,.18,Math.abs(y)),depth:Math.max(0,-y)};
}
function groundColor(seed:string,x:number,z:number){const f=floodedFields(seed,x,z),near=smooth(-10,40,z),c=new T.Color('#635a3d').lerp(new T.Color('#769353'),smooth(.015,.23,f.y));c.lerp(new T.Color('#b6b575'),f.patch*.25).multiplyScalar(.85+f.patch*.21);c.lerp(new T.Color('#284e4d'),near*.22);c.lerp(new T.Color('#8fae9b'),smooth(90,650,Math.hypot(x,z))*.65);return c;}
function sedgeGeometry(far=false){const pos:number[]=[],uv:number[]=[],ix:number[]=[],blades=far?3:7,steps=far?2:4;for(let j=0;j<blades;j++){const base=pos.length/3,a=j*2.399,w=far?.035:.065,h=.65+(j%4)*.16,lean=.18+(j%3)*.13;for(let k=0;k<=steps;k++)for(const side of [-1,1]){const t=k/steps,width=w*Math.sin(Math.PI*t)*side,reach=lean*t*t;pos.push(Math.cos(a)*reach-Math.sin(a)*width,h*t,Math.sin(a)*reach+Math.cos(a)*width);uv.push((side+1)/2,t);}for(let k=0;k<steps;k++){const a=base+k*2;ix.push(a,a+1,a+2,a+1,a+3,a+2);}}const g=new T.BufferGeometry();g.setAttribute('position',new T.Float32BufferAttribute(pos,3));g.setAttribute('uv',new T.Float32BufferAttribute(uv,2));g.setIndex(ix);g.computeVertexNormals();return g;}

  addLand(360,460,-170,170);addLand(2100,2000,-650,150,true);

  sun.position.set(-55,35,25);sun.color.set('#ffe3b0');sun.intensity=3;fill.color.set('#a5c7dc');fill.intensity=.4;scene.children.forEach(o=>{if(o instanceof T.HemisphereLight){o.color.set('#bfd9df');o.groundColor.set('#526d53');o.intensity=.9;}});scene.background=new T.Color('#c5d9da');scene.fog=new T.Fog('#b4cecb',130,1150);
  scene.userData.wave1={id:'flooded',version:'flooded-2',biome:recipe.ecoregion?.BIOME_NUM,ecoregion:recipe.ecoregion?.ECO_ID,seed:recipe.seed,treeCount:recipe.trees.length,sedgeInstances:counts,reflection:'Reflected sky lookup with subtle ripples; no planar geometry or tree reflection.',ecology:'Unnamed open-groundcover sedge/grass forms; no confirmed local taxon inferred.',layers:{season:'inherited/not inferred',weather:'fixed clear study',timeOfDay:'fixed warm directional study'},limitations:['Synthetic wetland microtopography, not surveyed hydrology.','Far marsh simplified; no tree islands inserted without source palette.']};
