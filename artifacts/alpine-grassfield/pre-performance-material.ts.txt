import * as T from 'three';
import {alpinePalettes,type AlpinePlan,type AlpinePaletteId} from './alpine-recipe';
/** Ground and grass share world-space pigments and detail; atlas colour never becomes a dark multiplier. */
export function meadowFieldMaterial(p:AlpinePlan,detail:T.Texture,atlas?:T.Texture,flower=false,instanced=!!atlas){
 const m=new T.MeshStandardMaterial({roughness:1,side:instanced?T.DoubleSide:T.FrontSide,alphaTest:atlas?.24:0});
 const u={meadowDetail:{value:detail},meadowAtlas:{value:atlas??detail},fieldBody:{value:new T.Color()},fieldShade:{value:new T.Color()},fieldSun:{value:new T.Color()},fieldAir:{value:new T.Color()},fieldRock:{value:new T.Color()}};
 const set=(id:AlpinePaletteId)=>{u.fieldBody.value.set(id==='B'?'#599550':id==='C'?'#78865a':'#6c9c49');u.fieldShade.value.set(id==='B'?'#2c5559':id==='C'?'#45536a':'#2c5550');u.fieldSun.value.set(id==='B'?'#c2d58e':id==='C'?'#e0c28c':'#d9d580');u.fieldAir.value.set(alpinePalettes[id].mountain);u.fieldRock.value.set(alpinePalettes[id].rock);};set(p.palette);m.userData.alpineMeadowPalette=set;
 m.onBeforeCompile=shader=>{Object.assign(shader.uniforms,u);shader.vertexShader=`varying vec3 fieldWorld;varying float fieldHeight;varying vec2 fieldUv;${instanced?'attribute vec3 meadowRootNormal;varying vec3 fieldRootNormal;':''}${atlas?'attribute float brushTile;varying vec2 fieldAtlasUv;':''}\n`+shader.vertexShader;
 shader.vertexShader=shader.vertexShader.replace('#include <begin_vertex>',`#include <begin_vertex>\nfieldWorld=(modelMatrix*${instanced?'instanceMatrix*':''}vec4(position,1.)).xyz;fieldUv=uv;${instanced?'fieldRootNormal=meadowRootNormal;':''}fieldHeight=${instanced?'fieldWorld.y-(modelMatrix*instanceMatrix*vec4(0.,0.,0.,1.)).y':'0.'};${atlas?'fieldAtlasUv=(vec2(mod(brushTile,2.),1.-floor(brushTile/2.))+uv)*.5;':''}`);
 shader.fragmentShader=`uniform sampler2D meadowDetail;uniform sampler2D meadowAtlas;uniform vec3 fieldBody,fieldShade,fieldSun,fieldAir,fieldRock;varying vec3 fieldWorld;varying float fieldHeight;varying vec2 fieldUv;${instanced?'varying vec3 fieldRootNormal;':''}${atlas?'varying vec2 fieldAtlasUv;':''}
 float fieldFlow(vec2 p){return .5+.22*sin(p.x*.049+p.y*.021+sin(p.y*.035)*1.4)+.16*sin(p.x*.083-p.y*.041+${p.phase.toFixed(5)})+.09*sin(p.y*.14+p.x*.037);}
 vec3 meadowPigment(vec2 p){float near=smoothstep(10.,43.,p.y),broken=.5+.5*sin((p.x+p.y*.42+7.*sin(p.y*.035+${p.phase.toFixed(5)}))*.09+sin(p.y*.026)*1.4);float opening=exp(-pow((p.x-2.-sin(p.y*.027)*7.)/25.,2.)-pow((p.y-4.)/81.,2.));float edge=smoothstep(5.,30.,abs(p.x))*(1.-smoothstep(55.,150.,-p.y));float shade=clamp(near*.58+edge*.35+broken*.28,0.,1.);float light=opening*(1.-shade*.5)*(.78+.22*(1.-broken))*(1.-near*.95);vec3 c=mix(fieldBody,fieldShade,shade*.78);c=mix(c,fieldSun,light*.82);c*=.90+.17*fieldFlow(p);float dist=length(p-vec2(${p.camera.position[0].toFixed(4)},${p.camera.position[2].toFixed(4)}));return mix(c,fieldAir,smoothstep(90.,650.,dist)*.48);}
 `+shader.fragmentShader;
 shader.fragmentShader=shader.fragmentShader.replace('#include <color_fragment>',`#include <color_fragment>\nfloat dist=distance(cameraPosition,fieldWorld);float detailFade=1.-smoothstep(30.,160.,dist);float ink=dot(texture2D(meadowDetail,fieldWorld.xz/6.).rgb,vec3(.2126,.7152,.0722));vec3 base=meadowPigment(fieldWorld.xz)*mix(1.,clamp(ink/.22,.78,1.25),.48*detailFade${instanced?'*(1.-smoothstep(.06,.30,fieldHeight))':''});diffuseColor.rgb=base;${atlas?'vec4 painted=texture2D(meadowAtlas,fieldAtlasUv);float h=smoothstep(.06,.30,fieldHeight);':''}${atlas?(flower?'vec3 flowerPaint=mix(vec3(dot(painted.rgb,vec3(.2126,.7152,.0722))),painted.rgb,.82)*mix(vec3(.65,.78,.85),vec3(1.02,1.,.9),clamp(dot(base,vec3(.25,.6,.15))*3.,0.,1.));diffuseColor.rgb=mix(base,flowerPaint,h);':'float stroke=clamp(dot(painted.rgb,vec3(.2126,.7152,.0722))/.43,.72,1.30);diffuseColor.rgb=base*mix(1.,stroke,.35*h);diffuseColor.rgb=mix(diffuseColor.rgb,fieldSun,smoothstep(.25,.9,fieldHeight)*.13);'):''}`);
 if(atlas){shader.fragmentShader=shader.fragmentShader.replace('#include <alphatest_fragment>',`float border=smoothstep(0.,.025,fieldUv.x)*smoothstep(0.,.025,1.-fieldUv.x)*smoothstep(0.,.025,1.-fieldUv.y);diffuseColor.a*=painted.a*border*(1.-smoothstep(${flower?'70.,130.':'35.,90.'},dist));\n#include <alphatest_fragment>`);}
 if(instanced)shader.fragmentShader=shader.fragmentShader.replace('#include <normal_fragment_begin>','#include <normal_fragment_begin>\nnormal=normalize(mat3(viewMatrix)*fieldRootNormal+normal*(.06*smoothstep(.06,.35,fieldHeight)));');
 };
 m.customProgramCacheKey=()=>`alpine-meadow-field-2-${!!atlas}-${flower}-${instanced}`;return m;
}
export function flowerFieldGeometry(){const pos:number[]=[],uv:number[]=[],ix:number[]=[];for(let face=0;face<2;face++){const a=(face-.5)*.7,co=Math.cos(a),si=Math.sin(a),base=pos.length/3;for(let j=0;j<=3;j++)for(let k=0;k<3;k++){const t=j/3,u=k/2,x=(u-.5)*1.25+t*t*.12,z=Math.sin(u*Math.PI)*.12+t*t*.1;pos.push(x*co-z*si,-.1+t,x*si+z*co);uv.push(u,t);}for(let j=0;j<3;j++)for(let k=0;k<2;k++){const a=base+j*3+k,b=a+3;ix.push(a,a+1,b,a+1,b+1,b);}}const g=new T.BufferGeometry();g.setAttribute('position',new T.Float32BufferAttribute(pos,3));g.setAttribute('uv',new T.Float32BufferAttribute(uv,2));g.setIndex(ix);g.computeVertexNormals();return g;}
