/** Real tapered, cupped blades; only the near batch carries sparse opaque seed heads. */
function physicalGrassGeometry(detail:boolean){
 const positions:number[]=[],normals:number[]=[],uvs:number[]=[],seed:number[]=[];
 const tri=(points:number[][],tex:number[][],part=0)=>{for(let i=0;i<3;i++){positions.push(...points[i]);uvs.push(...tex[i]);seed.push(part);normals.push(0,.8,.6);}};
 const blades=detail?5:3,steps=detail?4:2,cols=detail?2:1;

  const point=(u:number,t:number)=>{const w=(detail?.045:.055)*Math.pow(Math.max(0,Math.sin(t*Math.PI)),.6),x=(j-2)*.14+t*t*(.48+j*.04)+(u-.5)*w*2,z=Math.sin(j*2.4)*.17+Math.sin(u*Math.PI)*Math.sin(t*Math.PI)*.035;return [x*c-z*s,Math.sin(t*Math.PI*.57)*height,x*s+z*c];};
  for(let k=0;k<steps;k++)for(let col=0;col<cols;col++){const u=col/cols,v=(col+1)/cols,t=k/steps,b=(k+1)/steps;tri([point(u,t),point(v,t),point(u,b)],[[u,t],[v,t],[u,b]]);tri([point(v,t),point(v,b),point(u,b)],[[v,t],[v,b],[u,b]]);}

/** Closed triangular cross-sections follow a bowed centreline; roots are buried, tips taper to zero.
 * No atlas, transparency or per-blade shadow maps. All variation is deterministic local geometry. */
function dimensionalGrassGeometry(detail:boolean){
 const vertices:number[]=[],uv:number[]=[],seed:number[]=[],blades=detail?5:2,steps=detail?6:4;
 const triangle=(a:T.Vector3,b:T.Vector3,c:T.Vector3,ta:number,tb:number,tc:number,part=0)=>{for(const [v,t]of [[a,ta],[b,tb],[c,tc]] as [T.Vector3,number][]){vertices.push(...v.toArray());uv.push(.5,t);seed.push(part);}};
 for(let j=0;j<blades;j++){
  const az=(j-(blades-1)*.5)*.43+Math.sin(j*3.7)*.16,dir=new T.Vector3(Math.cos(az),0,Math.sin(az)),side=new T.Vector3(-dir.z,0,dir.x),h=.7+(j%3)*.16,reach=.45+(j%3)*.14;
  const ring=(t:number,k:number)=>{const center=dir.clone().multiplyScalar((j-(blades-1)*.5)*.19+reach*t*t).addScaledVector(side,Math.sin(j*2.1)*.24+.08*Math.sin(t*3.14+j));center.y=-.06+h*(t-.2*t*t*t);const width=(detail?.052:.039)*Math.pow(Math.max(0,Math.sin(t*Math.PI)),.65),theta=k*2*Math.PI/3+.4*t;return center.addScaledVector(side,Math.cos(theta)*width).add(new T.Vector3(0,Math.sin(theta)*width*.3,0));};
  for(let k=0;k<steps;k++)for(let side=0;side<3;side++){const a=k/steps,b=(k+1)/steps;triangle(ring(a,side),ring(a,side+1),ring(b,side),a,a,b);triangle(ring(a,side+1),ring(b,side+1),ring(b,side),a,b,b);}
 }
 if(detail){const stem=(t:number,k:number)=>new T.Vector3(.15+t*t*.25+Math.cos(k*2.094)*.008,t*1.18,Math.sin(k*2.094)*.008);for(let k=0;k<3;k++){triangle(stem(0,k),stem(0,k+1),stem(1,k),0,0,1,1);triangle(stem(0,k+1),stem(1,k+1),stem(1,k),0,1,1,1);}for(let j=0;j<4;j++){const y=.96+j*.055,c=new T.Vector3(.4,y,0);for(let k=0;k<3;k++)triangle(c.clone().add(new T.Vector3(Math.cos(k*2.094)*.026,-.025,Math.sin(k*2.094)*.026)),c.clone().add(new T.Vector3(Math.cos((k+1)*2.094)*.026,-.025,Math.sin((k+1)*2.094)*.026)),c.clone().add(new T.Vector3(.015,.06,0)),.8,.8,1,1);}}
 const g=new T.BufferGeometry();g.setAttribute('position',new T.Float32BufferAttribute(vertices,3));g.setAttribute('uv',new T.Float32BufferAttribute(uv,2));g.setAttribute('seedPart',new T.Float32BufferAttribute(seed,1));g.computeVertexNormals();return g;
}
/** Shared ground pigments meet smooth curved tuft surfaces: indexed normals, buried roots, no atlas or blade shadows. */

  const base=positions.length/3,az=(j-(blades-1)*.5)*.27+Math.sin(j*3.4)*.24,dir=new T.Vector3(Math.cos(az),0,Math.sin(az)),side=new T.Vector3(-dir.z,0,dir.x),h=(near?p.height:.7)*(.72+.28*(.5+.5*Math.sin(j*2.7))),lean=p.bend*(.7+.3*Math.cos(j*1.8)),root=side.clone().multiplyScalar(Math.sin(j*2.4)*p.spread*.55).addScaledVector(dir,Math.cos(j*4.1)*.18);
  for(let k=0;k<=steps;k++)for(let col=0;col<3;col++){
   const t=k/steps,u=col-1,width=p.width*(near?1:.8)*Math.pow(Math.max(0,Math.sin(Math.PI*t)),.7),center=root.clone().addScaledVector(dir,lean*t*t),v=center.addScaledVector(side,u*width);v.y=-.08+h*(id==='c'?Math.sin(t*Math.PI*(.65+.08*Math.sin(j*1.3))):(t-.17*t*t*t))+(1-u*u)*width*.75*Math.sin(t*Math.PI);positions.push(...v.toArray());uv.push(col/2,t);seed.push(0);
  }
  for(let k=0;k<steps;k++)for(let col=0;col<2;col++){const a=base+k*3+col,b=a+3;indices.push(a,a+1,b,a+1,b+1,b);}
 }
 const g=new T.BufferGeometry();g.setAttribute('position',new T.Float32BufferAttribute(positions,3));g.setAttribute('uv',new T.Float32BufferAttribute(uv,2));g.setAttribute('seedPart',new T.Float32BufferAttribute(seed,1));g.setIndex(indices);g.computeVertexNormals();return g;
}
/** Several gently bowed painterly surfaces form one tuft volume; atlas carries coherent brush masses. */
function brushGeometry(near:boolean){
 const positions:number[]=[],uv:number[]=[],seed:number[]=[],indices:number[]=[],faces=2,steps=near?4:2;
 for(let f=0;f<faces;f++){const start=positions.length/3,angle=.25+f*.42,co=Math.cos(angle),si=Math.sin(angle);for(let k=0;k<=steps;k++)for(let side=0;side<3;side++){const t=k/steps,u=side/2,x=(u-.5)*(near?2.5:2.1)+t*t*.5,z=Math.sin(u*Math.PI)*.22+t*t*.22;positions.push(x*co-z*si,-.16+t*(near?1.55:.7),x*si+z*co);uv.push(u,t);seed.push(0);}for(let k=0;k<steps;k++)for(let side=0;side<2;side++){const a=start+k*3+side,b=a+3;indices.push(a,a+1,b,a+1,b+1,b);}}
 const g=new T.BufferGeometry();g.setAttribute('position',new T.Float32BufferAttribute(positions,3));g.setAttribute('uv',new T.Float32BufferAttribute(uv,2));g.setAttribute('seedPart',new T.Float32BufferAttribute(seed,1));g.setIndex(indices);g.computeVertexNormals();return g;
}
function surfaceMaterial(recipe:SceneRecipe,role:number,map:T.Texture,contacts:ReturnType<typeof foregroundPlan>=[],golden=false,plains=false,fidelity=false,tuft?:string,brushAtlas?:T.Texture){
 const p=recipeDefinition.parameters,phase=(hashSeed(recipe.seed)%6283)/1000;
 // The existing renderer has antialias:false. A2C there writes partial canvas alpha without coverage.
 const m=new T.MeshStandardMaterial({map,roughness:1,color:'#ffffff',side:role===2?T.DoubleSide:T.FrontSide,alphaTest:role===2&&brushAtlas?.24:role===2&&!fidelity?.12:0});
 const c=(s:string)=>`vec3(${new T.Color(s).toArray().map(n=>n.toFixed(5)).join(',')})`;
 const shared=`
 float contactField(vec2 p){float weight=0.;${contacts.map(a=>`{vec2 q=(p-vec2(${a.x.toFixed(4)},${a.z.toFixed(4)}))/vec2(${(a.scale*2.6).toFixed(4)},${(a.scale*1.8).toFixed(4)});q+=vec2(sin(q.y*3.),sin(q.x*2.))*.16;weight=max(weight,exp(-dot(q,q)*2.));}`).join('')}return weight;}
 float sweep(vec2 p){return .5+.25*sin(p.y*.13+p.x*.032+sin(p.x*.085+${phase.toFixed(5)})*1.6)+.18*sin(p.x*.11-p.y*.043+${phase.toFixed(5)});}
 float clearing(vec2 p){p+=vec2(sin(p.y*.12)*3.,cos(p.x*.08)*2.);vec2 q=(p-vec2(2.,2.))/vec2(23.,29.);return exp(-dot(q,q));}
 vec3 pigment(vec2 p){float s=clamp(.5+(sweep(p)-.5)*${p.macroContrast.toFixed(4)},0.,1.),f=clearing(p);vec3 a=mix(${c(plains?'#D99B37':golden?'#9b914a':'#7f8a5b')},${c(plains?'#F4BD55':golden?'#f0b13a':'#e5ac46')},smoothstep(.1,.85,s));a=mix(a,${c(plains?'#FFE0A0':golden?'#f3cf6f':'#e9c16b')},f*.62);float nearShade=smoothstep(0.,24.,p.y)*(${golden?'.3+.7*cos(p.x*.11+p.y*.09+sin(p.y*.08))*.5+.35':'.65+.35*cos(p.x*.11+sin(p.y*.08))'})*(1.-f*.35);a=mix(a,${c(plains?'#675E43':golden?'#526e78':'#5c7976')},nearShade*${golden?'.6':'.9'});${golden?`vec2 left=(p-vec2(-9.,27.))/vec2(8.,12.),right=(p-vec2(12.,24.))/vec2(9.,10.);left.x+=sin(p.y*.12)*.18;float frame=max(exp(-dot(left,left)),exp(-dot(right,right))*.65);a=mix(a,${c(plains?'#675E43':'#466f78')},frame*.78);`:''}${fidelity?`float north=34.-p.y+sin(p.x*.045+p.y*.027)*5.;float progress=smoothstep(2.,30.,north);vec3 near=mix(${c('#6C603E')},${c('#99763C')},smoothstep(.15,.85,s));vec3 middle=mix(${c('#E9B650')},${c('#FFE0A0')},smoothstep(.12,.85,s)*.72);a=mix(near,middle,progress);float breakLight=smoothstep(.63,.96,sweep(p+vec2(11.,-3.)));a=mix(a,${c('#E9B650')},breakLight*(1.-progress)*.4);float rollShade=(1.-smoothstep(.24,.57,sweep(p+vec2(-8.,12.))))*.26;a=mix(a,${c('#6C603E')},rollShade*(1.-progress*.7));`:''}return a;}
 `;
 m.onBeforeCompile=shader=>{
  if(role===2&&brushAtlas){shader.uniforms.brushAtlas={value:brushAtlas};shader.vertexShader='attribute float brushTile;varying vec2 brushUv;varying vec2 brushLocal;\n'+shader.vertexShader;shader.vertexShader=shader.vertexShader.replace('#include <uv_vertex>','#include <uv_vertex>\nbrushLocal=uv;brushUv=(vec2(mod(brushTile,2.),floor(brushTile/2.))+uv)*.5;');shader.fragmentShader='uniform sampler2D brushAtlas;varying vec2 brushUv;varying vec2 brushLocal;\n'+shader.fragmentShader;}
  shader.vertexShader='varying vec3 recipeWorld;varying float fibreTip;'+(plains&&role===2?'attribute float seedPart;attribute float grassSeedPresent;':'')+'\n'+shader.vertexShader;
  const world=role===2?'modelMatrix*instanceMatrix*vec4(position,1.)':'modelMatrix*vec4(position,1.)';
  shader.vertexShader=shader.vertexShader.replace('#include <begin_vertex>',`#include <begin_vertex>\nrecipeWorld=(${world}).xyz;fibreTip=uv.y;${plains&&role===2?'if(seedPart>.5&&grassSeedPresent<.5)transformed=vec3(0.);':''}`);
  shader.vertexShader=shader.vertexShader.replace('#include <uv_vertex>',`#include <uv_vertex>\nvec3 recipeSource=(${world}).xyz;vMapUv=vec2(recipeSource.x*.85+recipeSource.z*.3,recipeSource.z*.8-recipeSource.x*.15)/${p.tileMeters.toFixed(4)};`);
  shader.fragmentShader=shared+'\nvarying vec3 recipeWorld;varying float fibreTip;\n'+shader.fragmentShader;
  shader.fragmentShader=shader.fragmentShader.replace('#include <color_fragment>',`#include <color_fragment>
   float d=distance(cameraPosition,recipeWorld);float fine=1.-smoothstep(25.,100.,d);vec3 base=pigment(recipeWorld.xz);${golden?`base=mix(base,${c('#c49e57')},smoothstep(65.,150.,d)*.3);`:''}${plains?`float air=length(recipeWorld.xz-vec2(0.,34.));base=mix(base,${c('#E7B65D')},smoothstep(12.,65.,air)*.4);base=mix(base,${c('#C4B48B')},smoothstep(55.,130.,air)*.6);base=mix(base,${c('#91A5A0')},smoothstep(110.,185.,air)*.6);`:''}float painted=clamp(dot(diffuseColor.rgb,vec3(.2126,.7152,.0722))/.36,.55,1.45);
   diffuseColor.rgb=mix(base,${c('#a58d65')},contactField(recipeWorld.xz)*.36)*mix(1.,painted,${p.textureContrast.toFixed(4)}*(.4+.6*fine));
   ${role===1?`diffuseColor.rgb=mix(${c('#738c8a')},${c('#9ba991')},smoothstep(-2.,3.,recipeWorld.y));`:role===2?(fidelity?`diffuseColor.rgb=mix(diffuseColor.rgb,${c(tuft?'#FFE0A0':'#E9B650')},smoothstep(${tuft?' .25,.95':'.35,.95'},fibreTip)*${brushAtlas?'.10':tuft?tuftProfile(tuft).highlight+(tuft==='c'?'*(1.-smoothstep(18.,65.,d))':''):'.12'});`:'diffuseColor.rgb*=mix(1.,1.04,fibreTip);diffuseColor.a*=smoothstep(0.,.12,fibreTip)*(1.-smoothstep(70.,120.,d));'):''}`);
  if(role===2&&brushAtlas)shader.fragmentShader=shader.fragmentShader.replace('#include <alphatest_fragment>',`vec4 brushPaint=texture2D(brushAtlas,brushUv);float border=smoothstep(0.,.025,brushLocal.x)*smoothstep(0.,.025,1.-brushLocal.x)*smoothstep(0.,.03,1.-brushLocal.y);diffuseColor.a*=brushPaint.a*border*(1.-smoothstep(28.,68.,distance(cameraPosition,recipeWorld)));float stroke=clamp(dot(brushPaint.rgb,vec3(.2126,.7152,.0722))/.43,.65,1.4);diffuseColor.rgb*=mix(1.,stroke,.36);vec3 strokeHue=brushPaint.rgb/max(.08,dot(brushPaint.rgb,vec3(.2126,.7152,.0722)));diffuseColor.rgb*=mix(vec3(1.),strokeHue,.3);\n#include <alphatest_fragment>`);
  if(role===2)shader.fragmentShader=shader.fragmentShader.replace('#include <normal_fragment_begin>',`#include <normal_fragment_begin>\nnormal=normalize(mat3(viewMatrix)*vec3(0.,1.,0.)+normal*${tuft?'(.03+.28*smoothstep(.2,.85,fibreTip))':plains?'mix(.10,.4,1.-smoothstep(8.,28.,distance(cameraPosition,recipeWorld)))':'.12'});`);
  shader.fragmentShader=shader.fragmentShader.replace('#include <opaque_fragment>',`${role!==1?`float warm=clearing(recipeWorld.xz)*.6+sweep(recipeWorld.xz)*.25;outgoingLight*=mix(vec3(.7,.9,1.15),vec3(1.3,1.15,.85),warm);`:`outgoingLight=mix(outgoingLight,${c('#8fa2a0')},.55);`}\n#include <opaque_fragment>`);
 };
 m.customProgramCacheKey=()=> 'reference-recipe-'+role+'-'+golden+'-'+plains+'-'+fidelity+'-'+tuft+'-'+!!brushAtlas+'-'+JSON.stringify(p);return m;
}
